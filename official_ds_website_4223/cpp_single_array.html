Question 2: Program to find the frequency of each element in the array 
#include<iostream.h>
#include<conio.h>

void main() {
    int a[10], n, i, j, count;

    clrscr();

    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter elements:\n";
    for(i = 0; i < n; i++) {
        cin >> a[i];
    }

    for(i = 0; i < n; i++) {
        count = 1;
        for(j = i + 1; j < n; j++) {
            if(a[i] == a[j]) {
                count++;
            }
        }

        int printed = 0;
        for(j = 0; j < i; j++) {
            if(a[i] == a[j]) {
                printed = 1;
                break;
            }
        }

        if(!printed) {
            cout << a[i] << " occurs " << count << " times\n";
        }
    }

    getch();
}

output:Enter number of elements: 5
Enter elements:
1 2 2 3 1
1 occurs 2 times
2 occurs 2 times
3 occurs 1 times


Question 3: Program to left rotate the elements of an array
