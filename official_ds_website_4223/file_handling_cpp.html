Question 1: Write a C++ program to create a new text file and write some text into it.

Question 8:write a c++ program to sort the linesof text filesof equal size 
#include<iostream.h>
#include<conio.h>
#include<vector.h>
#include<algorithm.h>
#include<fstream.h>
void main() {
    ifstream file("sample.txt");
    vector lines;
    string line;

    while(getline(file, line))
        lines.push_back(line);

    sort(lines.begin(), lines.end());

    for(string l : lines)
        cout << l << endl;

    file.close();
    getch();
}

output:File handling in C++ is easy.

Question 14: Read a CSV file and display in tabular form. 
#include<iostream.h>
#include<conio.h>
#include<sstream.h>
#include<fstream.h>
void main() {
    ifstream file("data.csv");
    string line;

    while(getline(file, line)) {
        stringstream ss(line);
        string item;

        while(getline(ss, item, ',')) {
            cout << item << "\t";
        }
        cout << endl;
    }

   getch();
}

Question 15:Calculate the average of numbers stored in a file 
#include<iostream.h>
#include<conio.h>
#include<fstream.h>
void main() {
    ifstream file("numbers.txt");

    int num, sum = 0, count = 0;

    while(file >> num) {
        sum += num;
        count++;
    }

    cout << "Average = " << (float)sum / count;

    getch();
}

output:Average = 45
