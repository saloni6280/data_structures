    scanf("%d",&d);
    printf("Enter the month: ");
    scanf("%d",&m);
    printf("Enter the year: ");
    scanf("%d",&y);

    if(y < 1 || m < 1 || m > 12 || d < 1)
        printf("Invalid date");
    else {
        if(m == 1 || m == 3 || m == 5 || m == 7 || m == 8 || m == 10 || m == 12)
            mdays = 31;
        else if(m == 4 || m == 6 || m == 9 || m == 11)
            mdays = 30;
        else if((y % 4 == 0 && y % 100 != 0) || (y % 400 == 0))
            mdays = 29;
        else
            mdays = 28;

        if(d <= mdays)
            printf("Valid date");
        else
            printf("Invalid date");
    }

    getch();
}

Question 23: Write a program to calculate area and perimeter of rectangle ?

#include<conio.h>

 {
    float l, b, area, peri;

    printf("Enter breadth: ");
