Question 1: CONSTRUCT BINARY TREE FROM PREORDER AND INORDER TRAVERSAL Preorder: Root, Left, Right Inorder: Left, Root, Right 
#include<stdio.h>
#include<stdlibh>
struct Node {
    int data;
    struct Node *left, *right;
};

int pre[100], in[100], index = 0;

struct Node* createNode(int data) {
    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

int search(int key, int start, int end) {
    for (int i = start; i <= end; i++) {
        if (in[i] == key)
            return i;
    }
    return -1;
}

struct Node* buildTree(int start, int end) {
    if (start > end)
        return NULL;

    int value = pre[index++];
    struct Node* root = createNode(value);

    int pos = search(value, start, end);

    if (pos == -1)
        return root;

    root->left = buildTree(start, pos - 1);
    root->right = buildTree(pos + 1, end);

    return root;
}

void postorder(struct Node* root) {
    if (root == NULL)
        return;

    postorder(root->left);
    postorder(root->right);

    printf("%d ", root->data);
}

int main() {
    int n;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter preorder traversal: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &pre[i]);

    printf("Enter inorder traversal: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &in[i]);

    struct Node* root = buildTree(0, n - 1);

    printf("Postorder traversal: ");
    postorder(root);

    return 0;
}

struct Node {
    int data;
    struct Node *left, *right;
};

    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    if (newNode == NULL) {
        printf("Memory allocation failed.");
        exit(1);
    }

    newNode->data = data;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;

int maxDepth(struct Node* root) {
    if (root == NULL)
        return 0;

    int leftDepth = maxDepth(root->left);
    int rightDepth = maxDepth(root->right);

    if (leftDepth > rightDepth)
        return leftDepth + 1;

    return rightDepth + 1;
}

    for (int i = 0; i < n; i++) {
        if (a[i] == -1)
            nodes[i] = NULL;
        else
            nodes[i] = createNode(a[i]);

    for (int i = 0; i < n; i++) {
        if (nodes[i] != NULL) {

            if (left < n)
                nodes[i]->left = nodes[left];

                nodes[i]->right = nodes[right];

int tree[MAX];

    for (int i = 0; i < MAX; i++)
        tree[i] = -1;
}

void insert(int value) {
    int i = 0;

    if (value < 0) {
        printf("Only non-negative keys are supported.\n");
        return;
    }

    while (i < MAX) {
        if (tree[i] == -1) {
            tree[i] = value;
            return;
        }

        if (value == tree[i]) {
            printf("Duplicate key ignored.\n");
            return;
        }

        if (value < tree[i])
            i = 2 * i + 1;
        else
            i = 2 * i + 2;
    }

    printf("Array capacity exceeded.\n");
}

int search(int value) {
    int i = 0;

    while (i < MAX && tree[i] != -1) {
        if (tree[i] == value)
            return i;

        if (value < tree[i])
            i = 2 * i + 1;
        else
            i = 2 * i + 2;
    }

    return -1;
}

void preorder(int i) {
    if (i >= MAX || tree[i] == -1)
        return;

    printf("%d ", tree[i]);
    preorder(2 * i + 1);
    preorder(2 * i + 2);
}

void inorder(int i) {
    if (i >= MAX || tree[i] == -1)
        return;

    inorder(2 * i + 1);
    printf("%d ", tree[i]);
    inorder(2 * i + 2);
}

void postorder(int i) {
    if (i >= MAX || tree[i] == -1)
        return;

    postorder(2 * i + 1);
    postorder(2 * i + 2);
    printf("%d ", tree[i]);
}

int minimum() {
    int i = 0;

    if (tree[0] == -1)
        return -1;

    while (2 * i + 1 < MAX &&
           tree[2 * i + 1] != -1) {
        i = 2 * i + 1;
    }

    return tree[i];
}

/* Delete a key and rebuild the BST from the remaining keys. */
void deleteKey(int value) {
    if (search(value) == -1) {
        printf("Key not found.\n");
        return;

    for (int i = 0; i < MAX; i++) {
        if (tree[i] != -1 && tree[i] != value)
            values[count++] = tree[i];
    }

    initialize();

    for (int i = 0; i < count; i++)
        insert(values[i]);

    printf("Key deleted successfully.\n");
}

struct Node* insert(struct Node* root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);

    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

struct Node* search(struct Node* root, int value) {
    while (root != NULL) {
        if (root->data == value)
            return root;

        if (value < root->data)
            root = root->left;
        else
            root = root->right;
    }

    return NULL;
}

struct Node* minimumNode(struct Node* root) {
    while (root != NULL && root->left != NULL)
        root = root->left;

    return root;
}

struct Node* deleteNode(struct Node* root, int value) {
    if (root == NULL)
        return NULL;

    if (value < root->data)
        root->left = deleteNode(root->left, value);

    else if (value > root->data)
        root->right = deleteNode(root->right, value);

    else {
        if (root->left == NULL) {
            struct Node* temp = root->right;
            free(root);
            return temp;
        }

        if (root->right == NULL) {
            struct Node* temp = root->left;
            free(root);
            return temp;
        }

        struct Node* temp = minimumNode(root->right);

        root->data = temp->data;

        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

Question 9: FIND INORDER PREDECESSOR AND INORDER SUCCESSOR OF A DYNAMICALLY REPRESENTED BINARY SEARCH TREE 
#include<stdio.h>
#include<stdlib.h>
struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* createNode(int value) {
    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    if (newNode == NULL)
        exit(1);

    newNode->data = value;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

struct Node* insert(struct Node* root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);

    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

void findPredecessorSuccessor(
    struct Node* root,
    int key,
    struct Node** predecessor,
    struct Node** successor
) {
    *predecessor = NULL;
    *successor = NULL;

    struct Node* current = root;

    while (current != NULL) {
        if (key < current->data) {
            *successor = current;
            current = current->left;
        }

        else if (key > current->data) {
            *predecessor = current;
            current = current->right;
        }

        else {
            if (current->left != NULL) {
                struct Node* temp = current->left;

                while (temp->right != NULL)
                    temp = temp->right;

                *predecessor = temp;
            }

            if (current->right != NULL) {
                struct Node* temp = current->right;

                while (temp->left != NULL)
                    temp = temp->left;

                *successor = temp;
            }

            return;
        }
    }
}

int main() {
    struct Node* root = NULL;
    struct Node* predecessor;
    struct Node* successor;

    int n, value, key;

    printf("Enter number of keys: ");
    scanf("%d", &n);

    printf("Enter BST keys: ");

    for (int i = 0; i < n; i++) {
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Enter target key: ");
    scanf("%d", &key);

    findPredecessorSuccessor(
        root, key, &predecessor, &successor
    );

    if (predecessor != NULL)
        printf("Inorder predecessor: %d\n",
               predecessor->data);
    else
        printf("Inorder predecessor: None\n");

    if (successor != NULL)
        printf("Inorder successor: %d\n",
               successor->data);

        printf("Inorder successor: None\n");

Question 10: CHECK WHETHER BINARY SEARCH TREE CONTAINS A DEAD END. 
#include<stdio.h>
#include<stdlib.h>
struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* createNode(int value) {
    struct Node* newNode =
        (struct Node*)malloc(sizeof(struct Node));

    if (newNode == NULL)
        exit(1);

    newNode->data = value;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

struct Node* insert(struct Node* root, int value) {
    if (root == NULL)
        return createNode(value);

    if (value < root->data)
        root->left = insert(root->left, value);

    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

int hasDeadEnd(struct Node* root, long long low,
               long long high) {
    if (root == NULL)
        return 0;

    if (root->left == NULL && root->right == NULL) {
        if (low == high)
            return 1;

        return 0;
    }

    return hasDeadEnd(root->left, low, root->data - 1LL)
        || hasDeadEnd(root->right, root->data + 1LL, high);
}

int main() {
    struct Node* root = NULL;

    int n, value;

    printf("Enter number of keys: ");
    scanf("%d", &n);

    printf("Enter positive BST keys: ");

    for (int i = 0; i < n; i++) {
        scanf("%d", &value);
        root = insert(root, value);
    }

    if (hasDeadEnd(root, 1, 1000000000LL))
        printf("BST contains a dead end.\n");
    else
        printf("BST does not contain a dead end.\n");

    return 0;
}

    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}
