Question 3: Define a class Matrix to represent matrices. Overload the * operator to multiply two matrices. 
#include<iostream.h>
#include<conio.h>

class Matrix {
private:
    int a[2][2];

public:
    void input() {
        cout << "Enter 4 elements:\n";
        for(int i=0;i<2;i++)
            for(int j=0;j<2;j++)
                cin >> a[i][j];
    }

    Matrix operator*(Matrix m) {
        Matrix temp;

        for(int i=0;i<2;i++) {
            for(int j=0;j<2;j++) {
                temp.a[i][j] = 0;
                for(int k=0;k<2;k++) {
                    temp.a[i][j] += a[i][k] * m.a[k][j];
                }
            }
        }

        return temp;
    }

    void display() {
        for(int i=0;i<2;i++) {
            for(int j=0;j<2;j++) {
                cout << a[i][j] << " ";
            }
            cout << endl;
        }
    }
};

void main() {
    Matrix m1, m2, m3;

    cout << "Enter first matrix\n";
    m1.input();

    cout << "Enter second matrix\n";
    m2.input();

    m3 = m1 * m2;

    cout << "Result Matrix\n";
    m3.display();

    getch();
}

3 4

2 0
1 2
Result Matrix

Question 4: Implement a class String to represent strings. Overload the + operator to concatenate two strings. 
#include<iostream.h>
#include<conio.h>
#include<cstream.h>

class String {
private:
    char str[50];

public:
    void input() {
        cout << "Enter string: ";
        cin >> str;
    }

    String operator+(String s) {
        String temp;
        strcpy(temp.str, str);
        strcat(temp.str, s.str);
        return temp;
    }

    void display() {
        cout << str << endl;
    }
};

void main() {
    String s1, s2, s3;

    s1.input();
    s2.input();

    s3 = s1 + s2;

    cout << "Concatenated String: ";
    s3.display();

    getch();
}

Enter string: World
Concatenated String: HelloWorld


Question 5:Define a class Date to represent dates. Overload the < operator to compare two dates. 
#include<iostream.h>
#include<conio.h>

class Date {
private:
    int d, m, y;

public:
    void input() {
        cout << "Enter day month year: ";
        cin >> d >> m >> y;
    }

    bool operator<(Date dt) {
        if (y < dt.y) return true;
        if (y == dt.y && m < dt.m) return true;
        if (y == dt.y && m == dt.m && d < dt.d) return true;
        return false;
    }
};

void main() {
    Date d1, d2;

    d1.input();
    d2.input();

    if(d1 < d2)
        cout << "First date is earlier";
    else
        cout << "Second date is earlier";

    getch();
}

output:Enter day month year: 10 5 2024
Enter day month year: 12 5 2024
First date is earlier


Question 6:Implement a class Money to represent money. Overload the + operator to add two amounts of money. 
#include<iostream.h>
#include<conio.h>
class Money {
private:
    int rupees, paise;

public:
    void input() {
        cout << "Enter rupees and paise: ";
        cin >> rupees >> paise;
    }

    Money operator+(Money m) {
        Money temp;

        temp.rupees = rupees + m.rupees;
        temp.paise = paise + m.paise;

        if(temp.paise >= 100) {
            temp.rupees += temp.paise / 100;
            temp.paise = temp.paise % 100;
        }

        return temp;
    }

    void display() {
        cout << "Rs " << rupees << "." << paise << endl;
    }
};

void main() {
    Money m1, m2, m3;

    m1.input();
    m2.input();

    m3 = m1 + m2;

    cout << "Total Money: ";
    m3.display();

    getch();
}

output:Enter rupees and paise: 50 75
Enter rupees and paise: 20 50
Total Money: Rs 71.25.


Question 7:Create a class Rectangle to represent rectangles. Overload the == operator to check if two rectangles are equal. 
#include<iostream.h>
#include<conio.h>
class Rectangle {
private:
    int length, width;

public:
    void input() {
        cout << "Enter length and width: ";
        cin >> length >> width;
    }

    bool operator==(Rectangle r) {
        return (length == r.length && width == r.width);
    }
};

void main() {
    Rectangle r1, r2;

    r1.input();
    r2.input();

    if(r1 == r2)
        cout << "Rectangles are equal";
    else
        cout << "Rectangles are not equal";

    getch();
}

Question 8: Define a class Time to represent time. Overload the << operator to display the time in HH:MM format. 
#include<iostream.h>
#include<conio.h>
class Time {
private:
    int hour, minute;

public:
    void input() {
        cout << "Enter hour and minute: ";
        cin >> hour >> minute;
    }

    friend ostream& operator<<(ostream &out, Time t) {
        out << t.hour << ":" << t.minute;
        return out;
    }
};

void main() {
    Time t;

    t.input();

    cout << "Time = " << t;

    getch();
}

output:Enter hour and minute: 10 45
Time = 10:45
