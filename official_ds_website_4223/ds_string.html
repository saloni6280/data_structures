            lines++;

        lines = 1;

    printf("Characters: %d\n", characters);
    printf("Words: %d\n", words);
    printf("Spaces: %d\n", spaces);
    printf("Lines: %d", lines);

    return 0;
}

Question 21: PRINT ALL THE PERMUTATION OF A STRING I.E. "ABCDE". 
#include<stdio.h>
#include<string.h>
void swap(char *a, char *b)
{
    char temp = *a;
    *a = *b;
    *b = temp;
}

void permutation(char str[], int start, int end)
{
    int i;

    if (start == end)
    {
        printf("%s\n", str);
        return;
    }

    for (i = start; i <= end; i++)
    {
        swap(&str[start], &str[i]);

        permutation(str, start + 1, end);

        swap(&str[start], &str[i]);
    }
}

int main()
{
    char str[] = "ABCDE";

    permutation(str, 0, strlen(str) - 1);

    return 0;
}

output:ABCDE
ABCED
ABDCE
ABDEC
ABECD
ABEDC
Total permutations = 120

Question 28: CREATE A DICTIONARY HAVING 10 WORDS AND THEIR MEANING WITH FOLLOWING OPERATIONS. [CREATE, DELETE, SEARCH]. 
#include<stdio.h>
#include<string.h>
struct Dictionary
{
    char word[30];
    char meaning[100];
};

int main()
{
    struct Dictionary d[20] = {
        {"Apple", "A fruit"},
        {"Book", "A collection of pages"},
        {"Cat", "A domestic animal"},
        {"Dog", "A pet animal"},
        {"Elephant", "A large animal"},
        {"Flower", "A part of a plant"},
        {"Garden", "A place for plants"},
        {"House", "A place to live"},
        {"Ink", "A writing liquid"},
        {"King", "A male ruler"}};

    int n = 10, choice, i, j, found;
    char word[30];

    do
    {
        printf("\n1. Create Word");
        printf("\n2. Delete Word");
        printf("\n3. Search Word");
        printf("\n4. Display Dictionary");
        printf("\n5. Exit");

        printf("\nEnter choice: ");
        scanf("%d", &choice);

        if (choice == 1)
        {
            if (n >= 20)
            {
                printf("Dictionary is full");
                continue;
            }

            printf("Enter word: ");
            scanf("%29s", d[n].word);

            printf("Enter meaning: ");
            scanf(" %99[^\n]", d[n].meaning);

            n++;

            printf("Word added successfully");
        }

        else if (choice == 2)
        {
            printf("Enter word to delete: ");
            scanf("%29s", word);

            found = 0;

            for (i = 0; i < n; i++)
            {
                if (strcmp(d[i].word, word) == 0)
                {
                    found = 1;

                    for (j = i; j < n - 1; j++)
                        d[j] = d[j + 1];

                    n--;

                    printf("Word deleted successfully");
                    break;
                }
            }

            if (found == 0)
                printf("Word not found");
        }

        else if (choice == 3)
        {
            printf("Enter word to search: ");
            scanf("%29s", word);

            found = 0;

            for (i = 0; i < n; i++)
            {
                if (strcmp(d[i].word, word) == 0)
                {
                    printf("Word: %s\n", d[i].word);
                    printf("Meaning: %s", d[i].meaning);

                    found = 1;
                    break;
                }
            }

            if (found == 0)
                printf("Word not found");
        }

        else if (choice == 4)
        {
            for (i = 0; i < n; i++)
            {
                printf("%s : %s\n",
                       d[i].word, d[i].meaning);
            }
        }

    } while (choice != 5);

    return 0;
}
