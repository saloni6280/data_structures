Question 1: Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form ? 
#include<stdio.h>
#Include<conio.h>
void main()
struct Employee
{
int id;
char name[50];
char designation[50];
};
int main()
{
struct Employee emp[50];
int i, n;
printf("Enter number of employees (max 50): ");
scanf("%d", &n);
for(i = 0; i < n; i++)
{
printf("\nEnter details of employee %d:\n", i + 1);
printf("Enter Employee ID: ");
scanf("%d", &emp[i].id);
printf("Enter Name: ");
scanf("%s", emp[i].name);  // single word name (can use fgets for full name)
printf("Enter Designation: ");
scanf("%s", emp[i].designation);
}
printf("\n---------------------------------------------------------\n");
printf("Emp.ID\t\tName\t\tDesignation\n");
printf("---------------------------------------------------------\n");
for(i = 0; i < n; i++)
{
printf("%d\t\t%-10s\t%-10s\n", emp[i].id, emp[i].name, emp[i].designation);
}
printf("---------------------------------------------------------\n");
getch();
}

Question 3:Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form ? 
#include<stdio.h>
#Include<conio.h>
void main()
struct Employee
{
int id;
char name[50];
char designation[50];
};
int main()
{
struct Employee emp[50];
int i, j, n;
printf("Enter number of employees (max 50): ");
scanf("%d", &n);
for(i = 0; i < n; i++)
{
printf("\nEnter details of employee %d:\n", i + 1);
printf("Enter Employee ID: ");
scanf("%d", &emp[i].id);
printf("Enter Name: ");
scanf("%s", emp[i].name);   // use fgets for full name
printf("Enter Designation: ");
scanf("%s", emp[i].designation);
}
printf("\n-------------------------------------------\n");
printf("Emp.ID\t\tName (character by character)\n");
printf("-------------------------------------------\n");
for(i = 0; i < n; i++)
{
printf("%d\t\t", emp[i].id);
for(j = 0; emp[i].name[j] != '\0'; j++)
{
printf("%c ", emp[i].name[j]);  // print each character with space
}
printf("\n");
}
printf("-------------------------------------------\n");
getch();
}
