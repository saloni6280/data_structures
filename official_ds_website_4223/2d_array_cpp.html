Question 12:Program to determine whether a given matrix is a sparse matrix
#include<iostream.h>
#include<conio.h>


void main() {
    clrscr();
    int a[10][10], r,c,zero=0;

    cin>>r>>c;

    for(int i=0;i>a[i][j];
            if(a[i][j]==0)
                zero++;
        }

    if(zero > (r*c)/2)
        cout<<"Sparse Matrix";
    else
        cout<<"Not Sparse";

    getch();
}

output: Sparse Matrix
