Question 1: Program to Check Anagram 
#include<iostream.h>
#include<conio.h>
#include<string.h>

    clrscr();
    char str1[50], str2[50];
    int freq1[256]={0}, freq2[256]={0};

    cout << "Enter first string: ";
    cin >> str1;
    cout << "Enter second string: ";
    cin >> str2;

    for(int i=0; str1[i]!='\0'; i++)
        freq1[str1[i]]++;
    for(int i=0; str2[i]!='\0'; i++)
        freq2[str2[i]]++;

    int flag = 1;
    for(int i=0; i<256; i++) {
        if(freq1[i] != freq2[i]) {
            flag = 0;
            break;
        }
    }

    if(flag)
        cout << "Strings are anagrams";
    else
        cout << "Strings are not anagrams";

    getch();
}

Question 17: Program to find all the permutations of a string 
#include<iostream.h>
#include<conio.h>
#include<string.h>
void permute(char str[], int l, int r) {
    if(l==r) {
        cout << str << "\n";
    } else {
        for(int i=l;i<=r;i++) {
            char temp = str[l]; str[l]=str[i]; str[i]=temp;
            permute(str, l+1, r);
            temp = str[l]; str[l]=str[i]; str[i]=temp; // backtrack
        }
    }
}

void main() {
    clrscr();
    char str[10];
    int len=0;
    cout << "Enter string (max 10 chars): ";
    cin >> str;
    while(str[len]!='\0') len++;
    cout << "Permutations:\n";
    permute(str,0,len-1);
    getch();
}

Question 39: Reverse String Word by Word

#include<conio.h>
#include<string.h>

    clrscr();
    char str[100], word[20][20];
    int i=0,j=0,n=0;
    cout << "Enter a string: ";
    cin.getline(str,100);

    while(str[i]!='\0') {
        if(str[i]!=' ') word[n][j++]=str[i];
        else if(j>0) { word[n][j]='\0'; n++; j=0; }
        i++;
    }
    if(j>0) { word[n][j]='\0'; n++; }

    cout << "Reversed words: ";
    for(i=n-1;i>=0;i--) cout << word[i] << " ";
    getch();
}

Question 48: Write a C++ method to check whether a string is a valid password.

#include<conio.h>
#include<string.h>

    clrscr();
    char str[50];
    int upper=0, lower=0, digit=0;
    cout << "Enter a password: ";
    cin >> str;

    for(int i=0; str[i]!='\0'; i++) {
        if(str[i]>='A' && str[i]<='Z') upper++;
        else if(str[i]>='a' && str[i]<='z') lower++;

    getch();
}
