Question 1: CREATE A STACK IMPLEMENTING THE OPERATIONS ON AN INTEGER ARRAY (PUSH, POP, PEEK, ISEMPTY, ISFULL) 
#include<stdio.h>
int top = -1;

int isempty()
{
    if (top == -1)
    {
        printf("stack is empty ! cannot delete\n");
        return 1;
    }
    else
    {
        printf("stack is not empty\n");
        return 0;
    }
}

int isfull()
{
    if (top == 4)
    {
        printf("stack is full ! cannot insert\n");
        return 1;
    }
    else
    {
        printf("stack is not full\n");
        return 0;
    }
}
void push(int stack[])
{
    if (isfull())
    {
        printf("stack is full cannot push\n");
    }
    else
    {
        int val;
        printf("enter element to push:");
        scanf("%d", &val);
        top++;
        stack[top] = val;
        printf("%d is pushed\n", stack[top]);
    }
}
void peek(int stack[])
{
    if (isempty())
    {
        printf("stack is empty\n");
    }
    else
    {
        printf("%d is top element in stack\n", stack[top]);
    }
}

void pop(int stack[])
{
    if (isempty())
    {
        printf("stack is empty cannot pop\n");
    }
    else
    {
        printf("%d is popped\n", stack[top]);
        top--;
    }
}

int main()
{
    int stack[5];
    int choice;
    printf("1:push \n2:pop\n3:peek\n4:isempty\n5:isfull\n6:exit\n");
    do
    {
        printf("enter your choice : ");
        scanf("%d", &choice);
        switch (choice)
        {

        case 1:
            push(stack);
            break;
        case 2:
            pop(stack);
            break;
        case 3:
            peek(stack);
            break;
        case 4:
            isempty();
            break;
        case 5:
            isfull();
            break;
        case 6:
            printf("program ended\n");
            break;
        default:
            printf("invalid number\n");
        }
    } while (choice != 6);
    return 0;

Question 3: CREATE A STACK IMPLEMENTING THE OPERATIONS ON AN INTEGER NODES (DYNAMICALLY) (PUSH, POP, PEEK, ISEMPTY, ISFULL) 
#include<stdio.h>
#include<stdlib.h>
struct node
{
    int data;
    struct node *next;
};
struct node *top = NULL;
int isfull()
{
    struct node *p;
    p = (struct node *)malloc(sizeof(struct node));
    if (p == NULL)
    {
        // printf("stack is full\n");
        return 1;
    }
    else
    {
        // printf("stack is not full\n");
        free(p);
        return 0;
    }
}
int isempty()
{
    if (top == NULL)
    {
        // printf("stack is empty\n");
        return 1;
    }
    else
    {
        // printf("stack is not empty\n");
        return 0;
    }
}
void push(int val)
{
    struct node *newnode;
    if (isfull())
    {
        printf("stack is full cannot push\n");
    }
    else
    {
        newnode = (struct node *)malloc(sizeof(struct node));
        newnode->data = val;
        newnode->next = top;
        top = newnode;
    }
    printf("%d is pushed\n", val);
}
void pop()
{
    struct node *temp;
    if (isempty())
    {
        printf("stack is empty cannot pop\n");
    }
    else
    {
        temp = top;
        top = top->next;
        printf("popped element is :%d\n", temp->data);
        free(temp);
    }
}
void peek()
{
    if (isempty())
    {
        printf("stack is empty cannot peek\n");
    }
    else
    {
        printf("%d is top element\n", top->data);
    }
}

void display()
{
    struct node *temp;
    if (isempty())
    {
        printf("stack  is empty cannot display\n");
    }
    else
    {
        temp = top;
        while (temp != NULL)
        {
            printf("elements in stack are: %d  ", temp->data);
            temp = temp->next;
        }
    }
}
int main()
{
    push(10);
    push(20);
    pop();
    peek();

    display();
    return 0;

top element is 10

            stack[top] = exp[i];

Question 7: CREATE TWO INTEGER STACKS IN A SINGLE ARRAY. 
#include<stdio.h>
int stack[10];
int top1 = -1, top2 = 10;
int i;

void push1(int x)
{
    if (top1 + 1 == top2)
        printf("Stack Overflow\n");
    else
        stack[++top1] = x;
}

void push2(int x)
{
    if (top1 + 1 == top2)
        printf("Stack Overflow\n");
    else
        stack[--top2] = x;
}

void display()
{
    printf("Stack 1: ");
    for (i = 0; i <= top1; i++)
        printf("%d ", stack[i]);

    printf("\nStack 2: ");
    for (i = 9; i >= top2; i--)
        printf("%d ", stack[i]);
}

int main()
{
    push1(10);
    push1(20);
    push1(30);

    push2(100);
    push2(200);
    push2(300);

    display();

    return 0;
}

output:stack 1:10 20 30
stack 2: 100 200 300


Question 8: SORT THE STACK IN ASCENDING ORDER USING INTEGER ARRAY. 
#include<stdio.h>
int stack[5], top = -1;
int i, j, temp;

int main()
{
    printf("Enter 5 elements:\n");

    for (i = 0; i < 5; i++)
    {
        scanf("%d", &stack[i]);
        top++;
    }

    for (i = 0; i < 5; i++)
    {
        for (j = i + 1; j < 5; j++)
        {
            if (stack[i] > stack[j])
            {
                temp = stack[i];
                stack[i] = stack[j];
                stack[j] = temp;
            }
        }
    }

    printf("Stack in ascending order:\n");

    for (i = 4; i >= 0; i--)
    {
        printf("%d\n", stack[i]);
    }

    return 0;
}

Question 10: ENTER THE INFIX EXPRESSION FROM USER AND CONVERT INTO POSTFIX EXPRESSION. 
#include<stdio.h>
#include<ctype.h>
char stack[50];
int top = -1;

void push(char x)
{
    stack[++top] = x;
}

char pop()
{
    return stack[top--];
}

int priority(char x)
{
    if (x == '+' || x == '-')
        return 1;
    if (x == '*' || x == '/')
        return 2;
    return 0;
}

void convert(char infix[])
{
    char x;
    int i;

    for (i = 0; infix[i] != '\0'; i++)
    {
        x = infix[i];

        if ((x >= 'A' && x <= 'Z') ||
            (x >= 'a' && x <= 'z') ||
            (x >= '0' && x <= '9'))
            printf("%c", x);

        else if (x == '(')
            push(x);

        else if (x == ')')
        {
            while (top != -1 && stack[top] != '(')
                printf("%c", pop());

            if (top != -1)
                pop();
        }

        else
        {
            while (top != -1 && priority(stack[top]) >= priority(x))
                printf("%c", pop());

            push(x);
        }
    }

    while (top != -1)
        printf("%c", pop());
}

int main()
{
    char infix[50];

    printf("Enter infix expression: ");
    scanf("%s", infix);

    printf("Postfix expression: ");
    convert(infix);

    return 0;
}

output:enter infix expression: a+b*c
postfix expression: abc*+


Question 11: CREATE A PROGRAM TO EVALUATE THE PREFIX EXPRESSION 
#include<stdio.h>

#include<sstring.h>

#include<ctype.h>

int stack[50];
int top = -1;

void push(int x)
{
    stack[++top] = x;
}

int pop()
{
    return stack[top--];
}

int evaluate(char prefix[])
{
    int i, a, b;
    int result;

    for (i = strlen(prefix) - 1; i >= 0; i--)
    {
        if (isdigit(prefix[i]))
            push(prefix[i] - '0');

        else
        {
            a = pop();
            b = pop();

            switch (prefix[i])
            {
            case '+':
                result = a + b;
                break;
            case '-':
                result = a - b;
                break;
            case '*':
                result = a * b;
                break;
            case '/':
                result = a / b;
                break;
            }

            push(result);
        }
    }

    return pop();
}

int main()
{
    char prefix[50];

    printf("Enter prefix expression: ");
    scanf("%s", prefix);

    printf("Result = %d", evaluate(prefix));

    return 0;
}

output:enter prefix expression: a+b*c
result=+a*bc
