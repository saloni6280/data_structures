Question 1: CALCULATE FACTORIAL OF A NUMBER USING RECURSION 
#include<stdio.h>
int factorial(int n)
{

    if (n == 0)
    {
        return 1;
    }
    else
    {
        return n * factorial(n - 1);
    }
}
int main()
{
    int res;
    res = factorial(3);
    printf(" factorial is %d", res);
    return 0;
}

output:factorial is 6


Question 2: CALCULATE FIBONACCI OF A NUMBER USING RECURSION 
#include<stdio.h>
int fab(int n){
    if(n==0){
        return 0;
    }
    if(n==1){
        return 1;
    }
    return fib(n - 1) + (n - 2);
}
int main(){
    int i;
    for (i=1;i<7;i++){
        printf("%d ", fib(i));
    }

    return 0; 
}

output:0 1 1 2 3 5 8

Question 6: FIND THE SMALLEST ELEMENT IN THE INTEGER ARRAY USING RECURSION 
#include<stdio.h>
int smallest(int a[], int n)
{
    int x;

    if (n == 1)
    {
        return a[0];
    }
    x = smallest(a, n - 1);
    if (a[n - 1] < x)
    {
        return a[n - 1];
    }
    else
    {
        return x;
    }
}

int main()
{
    int a[] = {10, 25, 5, 40, 15};
    printf("smallest is %d", smallest(a, 5));
    return 0;
}

Question 8: FIND THE ARITHMETIC MEAN OF AN INTEGER ARRAY USING RECURSION 
#include<stdio.h>
int sum(int a[],int n){
    if(n==0){
        return 0;
    }
    else{
        return a[n - 1] + sum(a, n - 1);
    }

}
int main(){
    int a[] = {10, 20, 30, 40, 50};
    int n = 5;
    float mean;
    mean = (float)sum(a, n) / n;
    printf("mean=%.2f", mean);
    return 0;
}

output:mean=30
