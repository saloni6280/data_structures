        {'1', '2', '3'},

        {'7', '8', '9'},

    char mark;
    while(1){
        for (i = 0; i < 3;i++){
            for (j = 0; j < 3;j++){
                printf("%c", a[i][j]);
            }
            printf("\n");
        }
        if(player==1){
            mark = 'X';
            
        }
        else{
            mark = 'O';

        }
        printf("player %d, enter position:", player);
        scanf("%d", &pos);
        row = (pos - 1) / 3;
        col = (pos - 1) % 3;
        if (pos < 1 || pos > 9 || a[row][col] == 'X' || a[row][col] == 'O'){
            printf("invalid position!\n");
            continue;
        }
        a[row][col] = mark;
        for (i = 0; i < 3;i++){
            if (a[i][0] == a[i][1] && a[i][1] == a[i][2]){
                printf("player %d wins!\n", player);
                return 0;

            {

                return 0;

        {
            printf("player %d wins!\n", player);

        player = 3 - player;
        }
