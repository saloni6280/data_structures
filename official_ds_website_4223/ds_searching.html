Question 2: ENTER A STRING FROM THE USER, ENTER A CHARACTER AND PERFORM LINEAR SEARCH ALGORITHM THROUGH RECURSION. 
#include<stdio.h>
int linear(char str[], char key, int i)
{
    if (str[i] == '\0')
        return -1;

    if (str[i] == key)
        return i;

    return linear(str, key, i + 1);
}

int main()
{
    char str[100], key;
    int result;

    printf("Enter a string: ");
    scanf("%s", str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    result = linear(str, key, 0);

    if (result == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", result + 1);

    return 0;
}

output:Enter a string: HELLO Enter character to search: L Character found at position 3

Question 5: ENTER A INTEGER ARRAY FROM THE USER, ENTER AN INTEGER VALUE AND PERFORM BINARY SEARCH ALGORITHM THROUGH ITERATION. 
#include<stdio.h>
int binary(int a[], int n, int key)
{
    int low = 0, high = n - 1, mid;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (a[mid] == key)
            return mid;

        else if (key > a[mid])
            low = mid + 1;

        else
            high = mid - 1;
    }

    return -1;
}

Question 6: ENTER A STRING FROM THE USER, ENTER A CHARACTER AND PERFORM BINARY SEARCH ALGORITHM THROUGH RECURSION. 
#include<stdio.h>
int binary(char str[], char key, int low, int high)
{
    int mid;

    if (low > high)
        return -1;

    mid = (low + high) / 2;

    if (str[mid] == key)
        return mid;

    else if (key > str[mid])
        return binary(str, key, mid + 1, high);

    else
        return binary(str, key, low, mid - 1);
}

int main()
{
    char str[100], key;
    int i, n = 0, result;

    printf("Enter sorted string: ");
    scanf("%s", str);

    while (str[n] != '\0')
        n++;

    printf("Enter character to search: ");
    scanf(" %c", &key);

    result = binary(str, key, 0, n - 1);

    if (result == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", result + 1);

    return 0;
}

Question 9: CALCULATE THE BEST AND WORST COMPLEXITY OF THE BINARY SEARCHING ALGORITHM WITH EXACT TIME TAKEN BY THE ALGORITHM ON INTEGER ARRAY. 
#include<stdio.h>
#include<time.h>
int binary(int a[], int n, int key, int *count)
{
    int low = 0, high = n - 1, mid;

    while (low <= high)
    {
        mid = (low + high) / 2;

        (*count)++;

        if (a[mid] == key)
            return mid;

        else if (key > a[mid])
            low = mid + 1;

        else
            high = mid - 1;
    }

    return -1;
}

int main()
{
    int a[100], n, i, key, result;
    int count = 0;

    clock_t start, end;
    double time_taken;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter sorted array elements:\n");

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &key);

    start = clock();

    result = binary(a, n, key, &count);

    end = clock();

    time_taken = (double)(end - start) / CLOCKS_PER_SEC;

    if (result == -1)
        printf("Element not found\n");
    else
        printf("Element found at position %d\n", result + 1);

    printf("Number of comparisons = %d\n", count);
    printf("Time taken = %f seconds\n", time_taken);

    printf("Best Complexity = O(1)\n");
    printf("Worst Complexity = O(log n)\n");

    return 0;
}

Best Complexity = O(1)
