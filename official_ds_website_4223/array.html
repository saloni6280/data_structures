Question 11: Enter five values of array and sort in ascending order ?
#include<stdio.h>
#include<conio.h>
#include<dos.h>
void main()
{
    int arr[5], i, j, temp;
    printf("Enter 5 elements:\n");
    for(i = 0; i < 5; i++)
        scanf("%d", &arr[i]);

    for(i = 0; i < 5; i++) {
        for(j = i + 1; j < 5; j++) {
            if(arr[i] > arr[j]) {
                temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }

    printf("Array in ascending order:\n");
    for(i = 0; i < 5; i++)
        printf("%d ", arr[i]);
}

output: Enter 5 elements : 5 4 3 2 1
Array in ascending order : 1 2 3 4 5


Question 12:Find maximum values of array ?

 {
    int arr[100], n, i, max;
    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    max = arr[0];
