#include<stdio.h>
void selectionSort(int a[], int n)
{
    int i, j, min, temp;

    for (i = 0; i < n - 1; i++)
    {
        min = i;

        // Find the smallest element
        for (j = i + 1; j < n; j++)
        {
            if (a[j] < a[min])
                min = j;
        }

        // Swap elements
        temp = a[i];
        a[i] = a[min];
        a[min] = temp;
    }
}

int main()
{
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    selectionSort(a, n);

    printf("Sorted array: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}

void insertionSort(int a[], int n)
{
    int i, j, key;

    for (i = 1; i < n; i++)
    {
        key = a[i];
        j = i - 1;

        // Move greater elements forward
        while (j >= 0 && a[j] > key)
        {
            a[j + 1] = a[j];
            j--;
        }

        a[j + 1] = key;
    }
}

#include<stdio.h>
void merge(int a[], int l, int m, int r)
{
    int temp[100], i = l, j = m + 1, k = 0;

    while (i <= m && j <= r)
    {
        if (a[i] < a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while (i <= m)
        temp[k++] = a[i++];

    while (j <= r)
        temp[k++] = a[j++];

    for (i = l, k = 0; i <= r; i++, k++)
        a[i] = temp[k];
}

void mergeSort(int a[], int n)
{

                m = n - 1;
            if (r >= n)
                r = n - 1;

            if (m < r)
                merge(a, l, m, r);
        }
    }
}

#include<stdio.h>
void quickSort(int a[], int n)
{
    int stack[100], top = -1;
    int low, high, i, j, pivot, temp;

    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0)
    {
        high = stack[top--];
        low = stack[top--];

        pivot = a[high];
        i = low - 1;

        for (j = low; j < high; j++)
        {
            if (a[j] < pivot)
            {
                i++;
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }

        temp = a[i + 1];
        a[i + 1] = a[high];
        a[high] = temp;

        j = i + 1;

        if (j - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = j - 1;
        }

        if (j + 1 < high)
        {
            stack[++top] = j + 1;
            stack[++top] = high;
        }
    }
}

{
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements: ");

#include<stdio.h>
int partition(int a[], int low, int high)
{
    int pivot = a[high];
    int i = low - 1, j, temp;

    for (j = low; j < high; j++)
    {
        if (a[j] < pivot)
        {
            i++;

            temp = a[i];
            a[i] = a[j];
            a[j] = temp;
        }
    }

    temp = a[i + 1];
    a[i + 1] = a[high];
    a[high] = temp;

    return i + 1;
}

void quickSort(int a[], int low, int high)
{
    int p;

    if (low < high)
    {
        p = partition(a, low, high);

        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

        if (j + 1 < high)

            stack[++top] = j + 1;
            stack[++top] = high;
        }
    }
}

#include<stdio.h>
#include<string.h>
int partition(char a[][50], int low, int high)
{
    char pivot[50], temp[50];
    int i = low - 1, j;

    strcpy(pivot, a[high]);

    for (j = low; j < high; j++)
    {
        if (strcmp(a[j], pivot) < 0)
        {
            i++;

            strcpy(temp, a[i]);
            strcpy(a[i], a[j]);
            strcpy(a[j], temp);
        }
    }

    strcpy(temp, a[i + 1]);
    strcpy(a[i + 1], a[high]);
    strcpy(a[high], temp);

    return i + 1;
}

void quickSort(char a[][50], int low, int high)
{
    int p;

    if (low < high)
    {
        p = partition(a, low, high);

        // Recursive calls
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}
