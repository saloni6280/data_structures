Question 2: CREATE THE GRAPH AS A ADJACENCY LIST WITH OPERATIONS SUCH AS INSERTION, DELETION? 
#include<stdio.h>
#include<stdlib.h>
#define MAX 100

struct Node
{
    int vertex;
    struct Node *next;
};

struct Node *adj[MAX];
int vertices;

void insertEdge(int u, int v)
{
    struct Node *newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));

    if (newNode == NULL)
    {
        printf("Memory allocation failed.\n");
        return;
    }

    newNode->vertex = v;
    newNode->next = adj[u];
    adj[u] = newNode;

    newNode = (struct Node *)malloc(sizeof(struct Node));

    if (newNode == NULL)
    {
        printf("Memory allocation failed.\n");
        return;
    }

    newNode->vertex = u;
    newNode->next = adj[v];
    adj[v] = newNode;

    printf("Edge inserted successfully.\n");
}

void deleteOneDirection(int u, int v)
{
    struct Node *temp = adj[u];
    struct Node *prev = NULL;

    while (temp != NULL)
    {
        if (temp->vertex == v)
        {
            if (prev == NULL)
                adj[u] = temp->next;
            else
                prev->next = temp->next;

            free(temp);
            return;
        }

        prev = temp;
        temp = temp->next;
    }
}

void deleteEdge(int u, int v)
{
    deleteOneDirection(u, v);
    deleteOneDirection(v, u);

    printf("Edge deleted if it existed.\n");
}

void displayGraph()
{
    int i;
    struct Node *temp;

    printf("\nAdjacency List:\n");

    for (i = 0; i < vertices; i++)
    {
        printf("%d: ", i);

        temp = adj[i];

        while (temp != NULL)
        {
            printf("%d -> ", temp->vertex);
            temp = temp->next;
        }

        printf("NULL\n");
    }
}

#include<stdio.h>
#define MAX 100

int graph[MAX][MAX];
int visited[MAX];
int vertices;

void DFS(int vertex)
{
    int i;

    visited[vertex] = 1;
    printf("%d ", vertex);

    for (i = 0; i < vertices; i++)
    {
        if (graph[vertex][i] == 1 && !visited[i])
        {
            DFS(i);
        }
    }
}

void BFS(int start)
{
    int front = 0, rear = 0;
    int vertex, i;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear)
