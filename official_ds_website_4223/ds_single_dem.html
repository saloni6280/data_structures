Question 8: Enter an array from user and find maximum and minimum value?

void findMinMax(int a[], int n)
{
    int i, min, max;
    min = max = a[0];

    for(i = 1; i < n; i++)
    {
        if(a[i] < min)
            min = a[i];
        if(a[i] > max)
            max = a[i];
    }
    printf("Minimum value = %d\n", min);
    printf("Maximum value = %d\n", max);
}

Question 9: Enter an array from user and find the k-th smallest value? 
#include<stdio.h>
#define MAX 100

void kthSmallest(int a[], int n, int k)
{
    int i, j, temp;
    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - 1 - i; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    if(k >= 1 && k <= n)
        printf("%dth smallest value = %d", k, a[k - 1]);
    else
        printf("Invalid value of k");
}

int main()
{
    int a[MAX], n, k, i;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);
    printf("Enter k: ");
    scanf("%d", &k);

    kthSmallest(a, n, k);
    return 0;
}

Question 13: Enter an array from user and count the duplicate values? 
#include<stdio.h>
#define MAX 100

void countDuplicates(int a[], int n)
{
    int i, j, count, already;
    int total = 0;

    for(i = 0; i < n; i++)
    {
        already = 0;
        for(j = 0; j < i; j++)
            if(a[i] == a[j])
                already = 1;

        if(already == 0)
        {
            count = 0;
            for(j = 0; j < n; j++)
                if(a[i] == a[j])
                    count++;

            if(count > 1)
            {
                printf("%d occurs %d times\n", a[i], count);
                total = total + count - 1;
            }
        }
    }
    printf("Total duplicate occurrences = %d", total);
}

int main()
{
    int a[MAX], n, i;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    countDuplicates(a, n);
    return 0;
}

Enter array elements: 2 4 2 5 4 2

Question 14: Enter an array from user and print all possible combinations of the elements? 
#include<stdio.h>
#define MAX 20

void combinations(int a[], int n)
{
    int i, j, total, bit;
    total = 1 << n;

    printf("All possible non-empty combinations:\n");
    for(i = 1; i < total; i++)
    {
        printf("{ ");
        for(j = 0; j < n; j++)
        {
            bit = 1 << j;
            if(i & bit)
                printf("%d ", a[j]);
        }
        printf("}\n");
    }
}

int main()
{
    int a[MAX], n, i;
    printf("Enter size of array (maximum 20): ");
    scanf("%d", &n);
    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    combinations(a, n);
    return 0;
}

Question 15: Enter an array from user and print all the palindrome numbers? 
#include<stdio.h>
#define MAX 100

int isPalindrome(int num)
{
    int original, reverse = 0, digit;
    original = num;

    if(num < 0)
        return 0;

    while(num > 0)
    {
        digit = num % 10;
        reverse = reverse * 10 + digit;
        num = num / 10;
    }

    if(original == reverse)
        return 1;
    else
        return 0;
}

void printPalindromes(int a[], int n)
{
    int i;
    printf("Palindrome numbers: ");
    for(i = 0; i < n; i++)
        if(isPalindrome(a[i]))
            printf("%d ", a[i]);
}

int main()
{
    int a[MAX], n, i;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printPalindromes(a, n);
    return 0;
}

Enter array elements: 121 34 55 123 7
Palindrome numbers: 121 55 7

Question 17: Merge the two sorted arrays into one array but in sorted order? 
#include<stdio.h>
#define MAX 100

void mergeSorted(int a[], int n, int b[], int m, int c[])
{
    int i = 0, j = 0, k = 0;

    while(i < n && j < m)
    {
        if(a[i] < b[j])
            c[k++] = a[i++];
        else
            c[k++] = b[j++];
    }

    while(i < n)
        c[k++] = a[i++];

    while(j < m)
        c[k++] = b[j++];
}

void display(int a[], int n)
{
    int i;
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);
}

int main()
{
    int a[MAX], b[MAX], c[2 * MAX];
    int n, m, i;
    printf("Enter size of first sorted array: ");
    scanf("%d", &n);
    printf("Enter first array in sorted order: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter size of second sorted array: ");
    scanf("%d", &m);
    printf("Enter second array in sorted order: ");
    for(i = 0; i < m; i++)
        scanf("%d", &b[i]);

    mergeSorted(a, n, b, m, c);
    printf("Merged sorted array: ");
    display(c, n + m);
    return 0;
}

output:Enter size of first sorted array: 3
Enter first array in sorted order: 1 4 7
Enter size of second sorted array: 3
Enter second array in sorted order: 2 3 8
Merged sorted array: 1 2 3 4 7 8


Question 18: Find the local minima in the array? (An element in an array is a local minima if it less than the element after it, and the element before it) 
#include<stdio.h>
#define MAX 100

void localMinima(int a[], int n)
{
    int i, found = 0;
    printf("Local minima: ");

    for(i = 1; i < n - 1; i++)
    {
        if(a[i] < a[i - 1] && a[i] < a[i + 1])
        {
            printf("%d ", a[i]);
            found = 1;
        }
    }

    if(found == 0)
        printf("No local minima found");
}

int main()
{
    int a[MAX], n, i;
    printf("Enter size of array: ");
    scanf("%d", &n);
    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    localMinima(a, n);
    return 0;
}

output:Enter size of array: 7
Enter array elements: 9 5 8 3 6 2 7
Local minima: 5 3 2
