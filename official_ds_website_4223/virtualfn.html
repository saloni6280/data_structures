Question 3: Write a C++ program to create an abstract class Shape with abstract methods calculateArea() and calculatePerimeter(). Create derived classes Circle and Triangle that extend the Shape class and implement the respective methods to calculate the area and perimeter of each shape. 
#include<iostream.h>
#include<conio.h>
void main() {
    clrscr();
    class Shape {
    public:
        virtual float calculateArea() = 0;
        virtual float calculatePerimeter() = 0;
    };

    class Circle : public Shape {
    public:
        float radius;
        Circle(float r) { radius=r; }
        float calculateArea() { return 3.14*radius*radius; }
        float calculatePerimeter() { return 2*3.14*radius; }
    };

    class Triangle : public Shape {
    public:
        float a,b,c;
        Triangle(float x,float y,float z) { a=x; b=y; c=z; }
        float calculateArea() { 
            float s=(a+b+c)/2;
            return sqrt(s*(s-a)*(s-b)*(s-c));
        }
        float calculatePerimeter() { return a+b+c; }
    };

    Shape *s;
    Circle c(5);
    Triangle t(3,4,5);

    s = &c;
    cout << "Circle Area: " << s->calculateArea() << "\n";
    cout << "Circle Perimeter: " << s->calculatePerimeter() << "\n";

    s = &t;
    cout << "Triangle Area: " << s->calculateArea() << "\n";
    cout << "Triangle Perimeter: " << s->calculatePerimeter();

    getch();
}

output: Circle Area: 78.5
Circle Perimeter: 31.4
Triangle Area: 6
Triangle Perimeter: 12

    BankAccount *b;
    SavingsAccount sa;
    CurrentAccount ca;

    b = &sa;
    b->deposit(1000);
    b->withdraw(500);

    b = &ca;
    b->deposit(2000);
    b->withdraw(2500);

Question 12: Write a C++ program to create an abstract class GeometricShape with abstract methods area() and perimeter(). Create derived classes Triangle and Square that extend the GeometricShape class and implement the respective methods to calculate the area and perimeter of each shape 
#include<iostream.h>
#include<conio.h>
void main() {
    clrscr();
    class GeometricShape {
    public:
        virtual double area()=0;
        virtual double perimeter()=0;
    };

    class Triangle : public GeometricShape {
    public:
        double a,b,c;
        Triangle(double x,double y,double z) { a=x; b=y; c=z; }
        double area() { 
            double s=(a+b+c)/2;
            return sqrt(s*(s-a)*(s-b)*(s-c));
        }
        double perimeter() { return a+b+c; }
    };

    class Square : public GeometricShape {
    public:
        double side;
        Square(double s) { side=s; }
        double area() { return side*side; }
        double perimeter() { return 4*side; }
    };

    GeometricShape *g;
    Triangle t(3,4,5);
    Square s(6);

    g = &t;
    cout << "Triangle Area: " << g->area() << "\n";
    cout << "Triangle Perimeter: " << g->perimeter() << "\n";

    g = &s;
    cout << "Square Area: " << g->area() << "\n";
    cout << "Square Perimeter: " << g->perimeter();

    getch();
}
