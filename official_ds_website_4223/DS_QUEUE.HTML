Question 3: IMPLEMENT THE OPERATIONS OF QUEUE DYNAMICALLY OF INTEGERS (INSERT, DELETE, PEEK, ISEMPTY, ISFULL) 
#include<stdio.h>
#include<stdlib.h>
struct node
{
    int data;
    struct node *next;
};
struct node *front = NULL;
struct node *rear = NULL;

int isempty()
{
    if (front == NULL && rear == NULL)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}
int isfull()
{
    struct node *p;
    p = (struct node *)malloc(sizeof(struct node));
    if (p == NULL)
    {
        return 1;
    }
    else
    {
        return 0;
    }
}
void enqueue(int x)
{
    struct node *newnode;
    if (isfull())
    {
        printf("\n queue is empty");
    }
    newnode = (struct node *)malloc(sizeof(struct node));
    newnode->data = x;
    newnode->next = NULL;
    if (front == NULL && rear == NULL)
    {
        front = rear = newnode;
    }
    else
    {
        rear->next = newnode;
        rear = newnode;
    }
    printf("\n%d is enqueued element", x);
}
void peek()
{
    if (isempty())
    {
        printf("\nqueue is empty");
    }
    else
    {
        printf("\n%d is front element", front->data);
    }
}

{
    struct node *temp;
    if (isempty())
    {
        printf("\nqueue is empty");
    }
    else if (front == rear)
    {
        printf("\n dequeued element is %d", front->data);
        front = rear = NULL;
    }
    else
    {
        temp = front;
        printf("\ndequeued element is %d", temp->data);
        front = front->next;
        free(temp);
    }
}

Question 7: IMPLEMENT THE OPERATIONS OF PRIORITY QUEUE IN AN INTEGER ARRAY (INSERT, DELETE, PEEK, ISEMPTY, ISFULL) 
#include<stdio.h>
#define SIZE 5

int queue[SIZE], priority[SIZE];
int count = 0;

int isEmpty()
{
    return count == 0;
}

int isFull()
{
    return count == SIZE;
}


void insert(int value, int p)
{
    if (isFull())
    {
        printf("Queue is Full\n");
        return;
    }

    queue[count] = value;
    priority[count] = p;
    count++;

    printf("Inserted successfully\n");
}


void delete()
{
    int i, pos = 0;

    if (isEmpty())
    {
        printf("Queue is Empty\n");
        return;
    }

    for (i = 1; i < count; i++)
    {
        if (priority[i] < priority[pos])
            pos = i;
    }

    printf("Deleted: %d\n", queue[pos]);

    for (i = pos; i < count - 1; i++)
    {
        queue[i] = queue[i + 1];
        priority[i] = priority[i + 1];
    }

    count--;
}


void peek()
{
    int i, pos = 0;

    if (isEmpty())
    {
        printf("Queue is Empty\n");
        return;
    }

    for (i = 1; i < count; i++)
    {
        if (priority[i] < priority[pos])
            pos = i;
    }

    printf("Front element: %d\n", queue[pos]);
}

Question 13: Length of the longest valid substring 
#include<stdio.h>
#include<string.h>
int longestValid(char str[])
{
    int stack[100], top = -1;
    int i, len, max = 0;

    len = strlen(str);

    // Initial index
    stack[++top] = -1;

    for (i = 0; i < len; i++)
    {
        if (str[i] == '(')
        {
            stack[++top] = i;
        }
        else
        {
            top--;

            if (top == -1)
                stack[++top] = i;
            else
            {
                if (i - stack[top] > max)
                    max = i - stack[top];
            }
        }
    }

    return max;
}

int main()
{
    char str[100];

    printf("Enter parentheses: ");
    scanf("%99s", str);

    printf("Longest valid length: %d",
           longestValid(str));

    return 0;
}

Question 16: Reverse First K elements of Queue 
#include<stdio.h>
void reverseK(int q[], int n, int k)
{
    int stack[100];
    int top = -1;
    int i, temp[100], index = 0;

    if (k < 0 || k > n)
    {
        printf("Invalid K value");
        return;
    }

    // Push first K elements into stack
    for (i = 0; i < k; i++)
        stack[++top] = q[i];

    // Pop into temporary array
    while (top != -1)
        temp[index++] = stack[top--];

    // Copy remaining queue elements
    for (i = k; i < n; i++)
        temp[index++] = q[i];

    printf("Queue after reversing: ");

    for (i = 0; i < n; i++)
        printf("%d ", temp[i]);
}

        scanf("%d", &q[i]);

Question 17: Queue Reversal 
#include<stdio.h>
void reverseQueue(int q[], int n)
{
    int stack[100];
    int top = -1;
    int i;

    // Push queue elements into stack
    for (i = 0; i < n; i++)
        stack[++top] = q[i];

    // Pop elements back into queue
    for (i = 0; i < n; i++)
        q[i] = stack[top--];

    printf("Reversed queue: ");

    for (i = 0; i < n; i++)
        printf("%d ", q[i]);
}
